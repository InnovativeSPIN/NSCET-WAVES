JOIN (
    SELECT reg_no,
           @grp := IF(@house = student_house, @grp + 1, 1) AS new_group,
           @house := student_house
    FROM registerationdb
    WHERE event_name = 'WARRIORS SHOWCASE'
    ORDER BY student_house, reg_no
) x ON r.reg_no = x.reg_no
SET r.grouped = x.new_group;
